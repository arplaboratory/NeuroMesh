cmake_minimum_required(VERSION 3.18)
project(neuromesh_onnx_stage VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(neuromesh_device_interface CONFIG REQUIRED)
if(NOT TARGET NeuroMesh::cuda_execution)
  message(FATAL_ERROR "Install device_interface with NEUROMESH_ENABLE_CUDA=ON")
endif()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(ONNXRuntime REQUIRED)
add_library(neuromesh_onnx_stage src/stage.cpp)
add_library(NeuroMesh::onnx_stage ALIAS neuromesh_onnx_stage)
set_target_properties(neuromesh_onnx_stage PROPERTIES EXPORT_NAME onnx_stage POSITION_INDEPENDENT_CODE ON)
target_compile_features(neuromesh_onnx_stage PUBLIC cxx_std_17)
target_include_directories(neuromesh_onnx_stage PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(neuromesh_onnx_stage PUBLIC NeuroMesh::cuda_execution PRIVATE ONNXRuntime::Runtime)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(neuromesh_onnx_stage PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS neuromesh_onnx_stage EXPORT NeuroMeshOnnxTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT NeuroMeshOnnxTargets NAMESPACE NeuroMesh:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neuromesh_onnx_stage)
configure_package_config_file(cmake/config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/neuromesh_onnx_stageConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neuromesh_onnx_stage)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/neuromesh_onnx_stageConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/neuromesh_onnx_stageConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/neuromesh_onnx_stageConfigVersion.cmake cmake/FindONNXRuntime.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neuromesh_onnx_stage)
option(NEUROMESH_ONNX_BOUNDED_ATTENTION "Build CUDA bounded-attention scheduling adapter" OFF)
if(NEUROMESH_ONNX_BOUNDED_ATTENTION)
  find_package(CUDAToolkit REQUIRED)
  add_library(neuromesh_onnx_bounded_attention SHARED src/bounded_attention.cpp)
  target_compile_features(neuromesh_onnx_bounded_attention PRIVATE cxx_std_17)
  target_link_libraries(neuromesh_onnx_bounded_attention PRIVATE ONNXRuntime::Runtime CUDA::cudart)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(neuromesh_onnx_bounded_attention PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  install(TARGETS neuromesh_onnx_bounded_attention LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
include(CTest)
if(BUILD_TESTING)
  set(NEUROMESH_ONNX_DYNAMIC_TEST_GRAPH "" CACHE FILEPATH "Symbolic [rows,2] identity graph")
  if(NEUROMESH_ONNX_DYNAMIC_TEST_GRAPH)
    add_executable(test_dynamic_stage tests/test_dynamic_stage.cpp)
    target_link_libraries(test_dynamic_stage PRIVATE NeuroMesh::onnx_stage)
    add_test(NAME onnx_symbolic_shape_specialization COMMAND test_dynamic_stage ${NEUROMESH_ONNX_DYNAMIC_TEST_GRAPH})
    set_tests_properties(onnx_symbolic_shape_specialization PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  endif()
  add_executable(test_onnx_stage tests/test_stage.cpp)
  target_link_libraries(test_onnx_stage PRIVATE NeuroMesh::onnx_stage)
  set(NEUROMESH_ONNX_TEST_GRAPH "" CACHE FILEPATH "Diagnostic prefix ONNX graph")
  set(NEUROMESH_ONNX_FIXTURE_DIR "" CACHE PATH "Generated native prefix fixture directory")
  if(NEUROMESH_ONNX_TEST_GRAPH AND NEUROMESH_ONNX_FIXTURE_DIR)
    foreach(source RANGE 0 1)
      add_test(NAME onnx_prefix_source_${source} COMMAND test_onnx_stage
        ${NEUROMESH_ONNX_TEST_GRAPH} ${NEUROMESH_ONNX_FIXTURE_DIR}/source_${source}_input.f32
        ${NEUROMESH_ONNX_FIXTURE_DIR}/source_${source}_expected.f32)
      set_tests_properties(onnx_prefix_source_${source} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    endforeach()
  else()
    message(STATUS "ONNX CUDA conformance tests not registered: graph/fixture paths required")
  endif()
endif()
