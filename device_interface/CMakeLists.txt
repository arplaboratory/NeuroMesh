cmake_minimum_required(VERSION 3.17)
project(neuromesh_device_interface VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
add_library(neuromesh_device_interface src/tensor.cpp src/completion.cpp src/tensor_pool.cpp)
target_link_libraries(neuromesh_device_interface PUBLIC Threads::Threads)
add_library(NeuroMesh::device_interface ALIAS neuromesh_device_interface)
set_target_properties(neuromesh_device_interface PROPERTIES EXPORT_NAME device_interface POSITION_INDEPENDENT_CODE ON)
target_compile_features(neuromesh_device_interface PUBLIC cxx_std_17)
target_include_directories(neuromesh_device_interface PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(neuromesh_device_interface PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS neuromesh_device_interface EXPORT NeuroMeshDeviceTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
option(NEUROMESH_ENABLE_CUDA "Build optional CUDA completion executor" OFF)
if(NEUROMESH_ENABLE_CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(neuromesh_cuda_execution src/cuda_executor.cpp src/cuda_upload.cpp src/cuda_download.cpp)
  add_library(NeuroMesh::cuda_execution ALIAS neuromesh_cuda_execution)
  set_target_properties(neuromesh_cuda_execution PROPERTIES EXPORT_NAME cuda_execution POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(neuromesh_cuda_execution PUBLIC NeuroMesh::device_interface CUDA::cudart)
  target_compile_features(neuromesh_cuda_execution PUBLIC cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(neuromesh_cuda_execution PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  install(TARGETS neuromesh_cuda_execution EXPORT NeuroMeshDeviceTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
install(EXPORT NeuroMeshDeviceTargets NAMESPACE NeuroMesh:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neuromesh_device_interface)
configure_package_config_file(cmake/config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/neuromesh_device_interfaceConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neuromesh_device_interface)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/neuromesh_device_interfaceConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/neuromesh_device_interfaceConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/neuromesh_device_interfaceConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/neuromesh_device_interface)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_device_tensor tests/test_tensor.cpp)
  target_link_libraries(test_device_tensor PRIVATE NeuroMesh::device_interface)
  add_test(NAME device_tensor_contract COMMAND test_device_tensor)
  add_executable(test_tensor_pool tests/test_tensor_pool.cpp)
  target_link_libraries(test_tensor_pool PRIVATE NeuroMesh::device_interface)
  add_test(NAME tensor_pool_contract COMMAND test_tensor_pool)
  add_executable(test_device_completion tests/test_completion.cpp)
  target_link_libraries(test_device_completion PRIVATE NeuroMesh::device_interface)
  add_test(NAME device_completion_contract COMMAND test_device_completion)
  set_tests_properties(device_completion_contract PROPERTIES TIMEOUT 10)
  if(NEUROMESH_ENABLE_CUDA)
    add_executable(test_cuda_executor tests/test_cuda_executor.cpp)
    target_link_libraries(test_cuda_executor PRIVATE NeuroMesh::cuda_execution)
    add_test(NAME cuda_executor_contract COMMAND test_cuda_executor)
    set_tests_properties(cuda_executor_contract PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    add_executable(test_cuda_upload tests/test_cuda_upload.cpp)
    target_link_libraries(test_cuda_upload PRIVATE NeuroMesh::cuda_execution)
    add_test(NAME cuda_upload_contract COMMAND test_cuda_upload)
    set_tests_properties(cuda_upload_contract PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
  endif()
endif()

install(FILES package.xml DESTINATION share/${PROJECT_NAME})
